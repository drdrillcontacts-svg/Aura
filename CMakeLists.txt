# JUCE is cloned automatically on first configure (internet access required
# once). JUCE 8.0.6 ships the VST3 SDK in-tree, so no submodules are needed.
# The clone lives in the build tree; delete it to force a re-clone.
set(AURA_JUCE_DIR "${CMAKE_BINARY_DIR}/juce-8.0.6")
set(AURA_JUCE_BINARY_DIR "${CMAKE_BINARY_DIR}/juce-build")

if(NOT EXISTS "${AURA_JUCE_DIR}/CMakeLists.txt")
    if(EXISTS "${AURA_JUCE_DIR}")
        file(REMOVE_RECURSE "${AURA_JUCE_DIR}")
    endif()
    execute_process(
        COMMAND git clone --depth 1 --branch 8.0.6
                https://github.com/juce-framework/JUCE.git "${AURA_JUCE_DIR}"
        RESULT_VARIABLE _aura_juce_clone_result
        OUTPUT_QUIET ERROR_QUIET)
    if(NOT _aura_juce_clone_result EQUAL 0)
        message(FATAL_ERROR
            "Could not clone JUCE 8.0.6 from https://github.com/juce-framework/JUCE.git - "
            "internet access is required once per machine/build directory.")
    endif()
endif()

# ----------------------------------------------------------------------------
# AURA writes its own JuceHeader.h (see aura_generate_juce_header below), so
# the 'juceaide' helper tool that JUCE normally builds at configure time is
# not needed. Building juceaide compiles JUCE's GUI modules, which requires
# X11/freetype development headers; skipping it also lets audio-only targets
# (the headless self-test) be configured on machines without those headers.
# (The patch is applied after the clone and before JUCE's CMake is run.)
#
# Exception: on native Windows juceaide IS built and used (it generates the
# .rc version/icon resources and works fine with MSVC).
# ----------------------------------------------------------------------------
if((CMAKE_SYSTEM_NAME STREQUAL "Linux" OR AURA_WINDOWS_CROSS_BUILD)
   AND EXISTS "${AURA_JUCE_DIR}/extras/Build/CMakeLists.txt")
    file(READ "${AURA_JUCE_DIR}/extras/Build/CMakeLists.txt" _aura_juce_build_list)
    if(_aura_juce_build_list MATCHES "add_subdirectory\\(juceaide\\)")
        string(REPLACE "add_subdirectory(juceaide)"
                       "# add_subdirectory(juceaide)  # disabled by AURA: JuceHeader.h is generated by aura_generate_juce_header()"
               _aura_juce_build_list_patched "${_aura_juce_build_list}")
        file(WRITE "${AURA_JUCE_DIR}/extras/Build/CMakeLists.txt" "${_aura_juce_build_list_patched}")
    endif()
    file(READ "${AURA_JUCE_DIR}/CMakeLists.txt" _aura_juce_root_list)
    if(_aura_juce_root_list MATCHES "get_target_property\\(JUCEAIDE_PATH juceaide IMPORTED_LOCATION\\)")
        string(REPLACE "get_target_property(JUCEAIDE_PATH juceaide IMPORTED_LOCATION)"
                       "if(TARGET juceaide)\n    get_target_property(JUCEAIDE_PATH juceaide IMPORTED_LOCATION)\nendif()"
               _aura_juce_root_patched "${_aura_juce_root_list}")
        file(WRITE "${AURA_JUCE_DIR}/CMakeLists.txt" "${_aura_juce_root_patched}")
    endif()
endif()

# ----------------------------------------------------------------------------
# Windows cross-compilation (cmake -DCMAKE_TOOLCHAIN_FILE=tools/cmake-windows-x86_64.cmake)
#
# The cross build uses zig-backed shims (tools/mingw-shim) as a mingw-w64
# substitute, so two host-side JUCE build steps need replacing:
#   1. The Windows resources .rc is normally generated by juceaide (disabled
#      here) -> write a minimal stub .rc instead (version metadata is omitted
#      in cross builds; the plugin is fully functional without it).
#   2. The VST3 moduleinfo.json is normally written by juce_vst3_helper, a
#      tool that would be built for Windows and thus could not run on the
#      build host -> copy the known-correct JSON from tools/moduleinfo-stub.json.
# ----------------------------------------------------------------------------
# The cross toolchain (tools/cmake-windows-x86_64.cmake) sets
# AURA_WINDOWS_CROSS_BUILD; a native Windows (MSVC) build must NOT take
# any of the steps below.
if(AURA_WINDOWS_CROSS_BUILD AND EXISTS "${AURA_JUCE_DIR}/extras/Build/CMake/JUCEUtils.cmake")
    file(READ "${AURA_JUCE_DIR}/extras/Build/CMake/JUCEUtils.cmake" _aura_juce_utils)

    # Note: \${...} below stays LITERAL (it must match the clone's own CMake
    # variable references); only ${CMAKE_*} paths are evaluated here.
    if(_aura_juce_utils MATCHES "COMMAND juce::juceaide rcfile")
        string(REPLACE "COMMAND juce::juceaide rcfile \"\${input_info_file}\" \"\${resource_rc_file}\""
                       "COMMAND \${CMAKE_COMMAND} -DAURA_RC_OUT=\"\${resource_rc_file}\" -P \"${CMAKE_CURRENT_LIST_DIR}/tools/aura_stub_rc.cmake\""
               _aura_juce_utils_patched "${_aura_juce_utils}")
        file(WRITE "${AURA_JUCE_DIR}/extras/Build/CMake/JUCEUtils.cmake" "${_aura_juce_utils_patched}")
        set(_aura_juce_utils "${_aura_juce_utils_patched}")
    endif()

    if(_aura_juce_utils MATCHES "COMMAND juce_vst3_helper")
        set(_aura_juce_utils_patched "${_aura_juce_utils}")
        string(REPLACE "COMMAND juce_vst3_helper"
                       "COMMAND \${CMAKE_COMMAND} -E copy \"${CMAKE_CURRENT_LIST_DIR}/tools/moduleinfo-stub.json\" \"\${ouput_path}\""
               _aura_juce_utils_patched "${_aura_juce_utils_patched}")
        string(REPLACE "            -create\n" "" _aura_juce_utils_patched "${_aura_juce_utils_patched}")
        string(REPLACE "            -version \"\${target_version_string}\"\n" "" _aura_juce_utils_patched "${_aura_juce_utils_patched}")
        string(REPLACE "            -path \"\${product}\"\n" "" _aura_juce_utils_patched "${_aura_juce_utils_patched}")
        string(REPLACE "            -output \"\${ouput_path}\"\n" "" _aura_juce_utils_patched "${_aura_juce_utils_patched}")
        file(WRITE "${AURA_JUCE_DIR}/extras/Build/CMake/JUCEUtils.cmake" "${_aura_juce_utils_patched}")
    endif()
endif()

add_subdirectory(${AURA_JUCE_DIR} ${AURA_JUCE_BINARY_DIR})

# ----------------------------------------------------------------------------
# Equivalent of juce_generate_juce_header(), but without the juceaide tool.
# Writes a static JuceHeader.h for the target (module list is fixed per
# target, so a configure-time write is sufficient).
# ----------------------------------------------------------------------------
function(aura_generate_juce_header target)
    get_target_property(generated_sources_directory ${target} JUCE_GENERATED_SOURCES_DIRECTORY)
    if(NOT generated_sources_directory)
        message(FATAL_ERROR "${target} has no JUCE generated sources directory")
    endif()

    set(juce_header "${generated_sources_directory}/JuceHeader.h")
    get_target_property(proj_name ${target} JUCE_PRODUCT_NAME)
    if(NOT proj_name)
        set(proj_name ${target})
    endif()
    get_target_property(company_name ${target} JUCE_COMPANY_NAME)
    if(NOT company_name)
        set(company_name "")
    endif()
    get_target_property(proj_version ${target} JUCE_VERSION)
    if(NOT proj_version)
        set(proj_version "1.0.0")
    endif()
    # version hex: MAJ<<16 | MIN<<8 | PATCH
    string(REGEX MATCH "^([0-9]+)\\.([0-9]+)\\.([0-9]+)" _m "${proj_version}")
    math(EXPR vhex "(((${CMAKE_MATCH_1}) << 16) | ((${CMAKE_MATCH_2}) << 8) | ${CMAKE_MATCH_3})")

    set(includes)
    foreach(mod IN LISTS ARGN)
        string(APPEND includes "#include <${mod}/${mod}.h>\n")
    endforeach()

    file(WRITE "${juce_header}"
"/*
    AURA JuceHeader - generated by aura_generate_juce_header() in CMakeLists.txt
    (a juceaide-free equivalent of juce_generate_juce_header).
*/
#pragma once

${includes}
using namespace juce;

namespace ProjectInfo
{
    const char* const  projectName    = \"${proj_name}\";
    const char* const  companyName    = \"${company_name}\";
    const char* const  versionString  = \"${proj_version}\";
    const int          versionNumber  =  0x${vhex};
}
")

    target_sources(${target} PRIVATE ${juce_header})
endfunction()

# Windows cross-build: msvcrt/ole32 gap fillers (long-double math,
# __isnan*, NaN/Inf sentinel data, CoCreateGuid) referenced by zig's CRT
# support but not provided by it, supplied from tools/win-gaps (see file
# headers for rationale).
#
# CMake in this setup silently drops plain .s sources (the ASM language
# cannot be compiler-detected against the zig cc shim), so the sentinel
# data object is pre-assembled at configure time into the build tree.
# Cross-build only: on native Windows these symbols come from the real
# msvcrt/ole32 and the -l import flags are meaningless to link.exe.
if(AURA_WINDOWS_CROSS_BUILD)
    set(_aura_gaps_s   "${CMAKE_CURRENT_SOURCE_DIR}/tools/win-gaps/aura_crt_gaps_data.s")
    set(_aura_gaps_obj "${CMAKE_BINARY_DIR}/aura_crt_gaps_data.obj")
    if(NOT EXISTS "${_aura_gaps_obj}" OR "${_aura_gaps_s}" IS_NEWER_THAN "${_aura_gaps_obj}")
        execute_process(
            COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/tools/mingw-shim/x86_64-w64-mingw32-cc"
                    -target x86_64-windows-gnu -x assembler
                    -c "${_aura_gaps_s}" -o "${_aura_gaps_obj}"
            RESULT_VARIABLE _aura_gaps_rc)
        if(NOT _aura_gaps_rc EQUAL 0)
            message(FATAL_ERROR "Failed to assemble aura_crt_gaps_data.s")
        endif()
    endif()
    target_sources(AURA PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/tools/win-gaps/aura_crt_gaps.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/tools/win-gaps/aura_uuid_gaps.cpp"
        "${_aura_gaps_obj}")

    # System DLL import libraries (real import thunks, generated by the
    # zig toolchain on demand) so the final link resolves the Win32 APIs
    # the JUCE modules reference.
    set(_aura_win_import_libs
        -lgdi32 -luser32 -lcomdlg32 -lshlwapi -ldwmapi -lole32 -loleaut32
        -ld2d1 -ld3d11 -ldxgi -ldcomp -lversion -limm32 -lwininet
        -ldbghelp -lwinmm -lws2_32
        "${CMAKE_BINARY_DIR}/win-imports/user32_supp.lib")
    target_link_libraries(AURA_VST3 PRIVATE ${_aura_win_import_libs})
    target_link_libraries(AURA_Standalone PRIVATE ${_aura_win_import_libs})
endif()

# ----------------------------------------------------------------------------
# Linux headless-build support
#
# JUCE's GUI modules need the X11 development headers (libx11-dev) to
# compile on Linux. On machines without them (headless CI / sandboxes), we
# fall back to the minimal compile-only stub headers in
# third_party/x11-stubs (JUCE loads X11 via dlopen at runtime, so only the
# declarations are needed at compile time). When the real headers are
# installed, they are used instead and the stubs are never included.
# ----------------------------------------------------------------------------
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_path(AURA_X11_XLIB_HEADER X11/Xlib.h)
    find_path(AURA_REAL_FT2BUILD_HEADER ft2build.h PATHS /usr/include/freetype2 /usr/local/include/freetype2)

    # JUCE's X11 code passes CopyFromParent (an int) where X11 expects a Visual*
    # (and vice versa); allow these implicit conversions in the two JUCE module
    # TUs that contain that code.
    set_source_files_properties(
        "${AURA_JUCE_DIR}/modules/juce_gui_basics/juce_gui_basics.cpp"
        "${AURA_JUCE_DIR}/modules/juce_gui_extra/juce_gui_extra.cpp"
        PROPERTIES COMPILE_OPTIONS "-fpermissive")

    foreach(_aura_target AURA AURA_SelfTest)
        # Compile-only stub headers (core X11 + FreeType). JUCE loads the real
        # libraries at runtime, so the declarations are only needed at compile
        # time; real development headers take precedence when installed.
        if(NOT AURA_X11_XLIB_HEADER)
            target_include_directories(${_aura_target} PRIVATE
                ${CMAKE_CURRENT_SOURCE_DIR}/third_party/x11-stubs)
        endif()
        if(NOT AURA_REAL_FT2BUILD_HEADER)
            target_include_directories(${_aura_target} PRIVATE
                ${CMAKE_CURRENT_SOURCE_DIR}/third_party/ft-stubs)
        endif()
        if(NOT AURA_X11_XLIB_HEADER)
            # The stubs cover core X11 only, so JUCE's X extension support
            # (Xrandr/Xinerama/XShm/XCursor) is disabled for these builds.
            target_compile_definitions(${_aura_target} PRIVATE
                JUCE_USE_XRANDR=0
                JUCE_USE_XINERAMA=0
                JUCE_USE_XSHM=0
                JUCE_USE_XCURSOR=0)
            message(STATUS "AURA: X11 dev headers not found - using compile-only stubs from third_party/x11-stubs (Linux binaries are for CI verification only)")
        endif()
        # fontconfig is only used for font fallback; keep it off so no
        # fontconfig development headers are required.
        target_compile_definitions(${_aura_target} PRIVATE JUCE_USE_FONTCONFIG=0)

        # No libasound2-dev / libjack dev packages on headless CI. JUCE documents
        # JUCE_ALSA / JUCE_JACK = 0 as the supported way to build without them;
        # plugin formats (VST3) do not use a system audio driver, so only the
        # standalone app's audio I/O is unavailable (irrelevant headless).
        find_path(AURA_ALSA_HEADER alsa/asoundlib.h)
        if(NOT AURA_ALSA_HEADER)
            target_compile_definitions(${_aura_target} PRIVATE JUCE_ALSA=0 JUCE_JACK=0)
        endif()

        if(NOT AURA_REAL_FT2BUILD_HEADER)
            # No system FreeType: provide link-time no-op definitions for the
            # FT_* entry points referenced by JUCE's bundled harfbuzz sources
            # (and the X11 dlsym fallbacks). Font loading then degrades
            # gracefully to an empty font list; audio is unaffected.
            target_sources(${_aura_target} PRIVATE
                ${CMAKE_CURRENT_SOURCE_DIR}/third_party/ft-stubs/stub-symbols.cpp)
        endif()
    endforeach()
endif()
